# --- PostgreSQL Installation on Ubuntu ---
# --- 1. Update Package Lists ---
# Ensure the system has the latest list of available software packages
sudo apt update

# --- 2. Install PostgreSQL ---
# Install the core server and useful additional modules (contrib)
sudo apt install postgresql postgresql-contrib

# --- 3. Manage the Service ---
# Start the database server immediately
sudo systemctl start postgresql

# Enable the server to start automatically on system boot
sudo systemctl enable postgresql

# Check the current status (should be "active/running")
sudo systemctl status postgresql

-- Example SQL Code Snippets --
-- Create a new database named 'CompanyDB'
CREATE DATABASE CompanyDB
WITH 
    OWNER = db_admin,          -- Specifies the owner of the database
    ENCODING = 'UTF8',         -- Sets the character encoding to UTF-8
    CONNECTION LIMIT = 25;     -- Limits the number of concurrent connections to 25;

CREATE TABLE Employees (
    EmployeeID INT AUTO_INCREMENT PRIMARY KEY, -- Unique identifier, auto-generated
    FirstName VARCHAR(50) NOT NULL,            -- This field cannot be empty
    LastName VARCHAR(50) NOT NULL,
    Email VARCHAR(100) UNIQUE,                 -- Ensures no duplicate email addresses
    BirthDate DATE,
    Salary DECIMAL(10, 2),                     -- Format for currency (e.g., 50000.00)
    Department VARCHAR(50) DEFAULT 'General'   -- Default value if none is provided
);

-- Create an index on the 'LastName' column to speed up search queries
CREATE INDEX idx_lastname 
ON Employees (LastName);

-- Create a view that shows only contact info (hides sensitive data like salary)
CREATE VIEW EmployeeContactList AS
SELECT FirstName, LastName, Email
FROM Employees;

-- Create or replace the procedure if it already exists to avoid errors (PostgreSQL syntax)
CREATE OR REPLACE PROCEDURE IncreaseSalary(
    p_EmployeeID INT, 
    p_Percentage DECIMAL
)
LANGUAGE plpgsql
AS $$
BEGIN
    -- Update the salary for the specific employee
    -- The calculation increases the current salary by the given percentage
    UPDATE Employees
    SET Salary = Salary * (1 + (p_Percentage / 100))
    WHERE EmployeeID = p_EmployeeID;

    -- Optional: Commit the transaction immediately (unique to Procedures in Postgres)
    COMMIT;
END;
$$;

-- Create or replace the procedure to allow overwriting without dropping it first (Oracle syntax)
CREATE OR REPLACE PROCEDURE IncreaseSalary (
    p_EmployeeID IN NUMBER,    -- Oracle prefers NUMBER instead of INT/DECIMAL
    p_Percentage IN NUMBER
)
AS
BEGIN
    -- Update the salary calculation
    -- Logic remains the same: Current Salary * (1 + Percentage/100)
    UPDATE Employees
    SET Salary = Salary * (1 + (p_Percentage / 100))
    WHERE EmployeeID = p_EmployeeID;

    -- In Oracle, you often commit explicitly to make changes permanent
    COMMIT;
END IncreaseSalary;
/

TRUNCATE OPERATIONS (TRUNCATE TABLE)
---------------------------------------------------------------------------------------------------------------
/* PostgreSQL Syntax */

-- 1. Simple Truncate: Clears data, keeps space allocated vaguely similar to before
TRUNCATE TABLE Logs;

-- 2. Reset Identity: Deletes data AND resets the auto-increment counter to 1
TRUNCATE TABLE Products RESTART IDENTITY;

-- 3. Cascade: The "Nuclear Option"
-- Deletes data in this table AND automatically deletes data in all tables 
-- that reference this table via Foreign Keys. Use with extreme caution!
TRUNCATE TABLE Departments CASCADE;

/* Oracle Syntax */

-- 1. Standard Truncate (implicitly performs DROP STORAGE)
-- Frees up the disk space for other tables to use
TRUNCATE TABLE Logs;

-- 2. Keep Space Allocated (Performance optimization)
-- Deletes rows but keeps the storage block reserved for this table.
-- Faster for subsequent INSERTs because no new space allocation is needed.
TRUNCATE TABLE Logs REUSE STORAGE;

-- 3. Cascade (Since Oracle 12c)
-- Recursively truncates child tables connected via Foreign Keys (ON DELETE CASCADE must be set)
TRUNCATE TABLE Departments CASCADE;

RENAME COLUMN Price TO UnitCost;

/* Oracle */
-- Requires Oracle 9i Release 2 or higher (standard now)
ALTER TABLE Inventory

SELECT STATEMENT (SELECT, FROM, WHERE, GROUP BY, HAVING, ORDER BY, JOINS, SUBQUERIES, SET OPERATIONS)
------------------------------------------------------------------------------------------------------
SELECT FirstName, LastName, Salary, Department
FROM Employees
WHERE Department = 'IT' 
  AND Salary > 50000
ORDER BY Salary DESC; -- Sorts from highest to lowest

SELECT Department, AVG(Salary) as AvgSalary, COUNT(*) as EmployeeCount
FROM Employees
GROUP BY Department
HAVING AVG(Salary) > 60000; -- Filters groups with average salary over 60,000

SELECT 
    e.FirstName, 
    e.LastName, 
    d.DepartmentName
FROM Employees e                  -- 'e' is an alias for Employees
INNER JOIN Departments d          -- 'd' is an alias for Departments
    ON e.DepartmentID = d.DepartmentID;

 SELECT FirstName, Salary
FROM Employees
WHERE Salary > (
    SELECT AVG(Salary) FROM Employees
); -- Finds employees earning above average salary

-- Returns a list of ALL names from both tables
-- UNION removes duplicates, UNION ALL keeps them (and is faster)
SELECT LastName FROM Current_Employees
UNION ALL
SELECT LastName FROM Archived_Employees;

-- Who is in the archive BUT somehow still listed as active?
SELECT EmployeeID FROM Current_Employees
INTERSECT
SELECT EmployeeID FROM Archived_Employees;

/* Oracle specific */
SELECT ProductID FROM Inventory
MINUS
SELECT ProductID FROM Sales; -- Items that have never been sold

/* PostgreSQL specific */
SELECT ProductID FROM Inventory
EXCEPT
SELECT ProductID FROM Sales; -- Items that have never been sold

JOINS (INNER JOIN, LEFT JOIN, RIGHT JOIN, FULL OUTER JOIN, CROSS JOIN, SELF JOIN)
---------------------------------------------------------------------------------------------------------------

/* Type: INNER JOIN
   Logic: Returns only rows when there is a match in BOTH tables.
   Result: Employees without a department are excluded. 
           Departments with no employees are excluded.
*/
SELECT e.LastName, d.DepartmentName
FROM Employees e
INNER JOIN Departments d 
    ON e.DepartmentID = d.DepartmentID;

/* Type: LEFT OUTER JOIN
   Logic: Returns ALL records from the LEFT table (Employees), and the matched records from the right table.
   Result: If an employee has no department, they are still listed, but DepartmentName will be NULL.
*/
SELECT e.LastName, d.DepartmentName
FROM Employees e
LEFT JOIN Departments d 
    ON e.DepartmentID = d.DepartmentID;  

/* Type: RIGHT OUTER JOIN
   Logic: Returns ALL records from the RIGHT table (Departments), and the matched records from the left table.
   Result: If a department exists (e.g., 'Innovation Lab') but has no employees, it is listed.
           The employee columns (LastName) will be NULL for that row.
*/
SELECT e.LastName, d.DepartmentName
FROM Employees e
RIGHT JOIN Departments d 
    ON e.DepartmentID = d.DepartmentID;

/* Type: FULL OUTER JOIN
   Logic: Returns all records when there is a match in EITHER left or right table.
   Result: Combines the results of both LEFT and RIGHT joins.
           - You see employees without departments.
           - You see departments without employees.
           - NULLs appear wherever data is missing on the other side.
*/
SELECT e.LastName, d.DepartmentName
FROM Employees e
FULL OUTER JOIN Departments d 
    ON e.DepartmentID = d.DepartmentID;

/* Type: CROSS JOIN (Cartesian Product)
   Logic: Returns the combination of EVERY row from the first table with EVERY row from the second table.
   Result: If Employees has 100 rows and Departments has 10 rows, the result is 1,000 rows.
   Warning: Can be very slow on large tables!
*/
SELECT e.LastName, d.DepartmentName
FROM Employees e
CROSS JOIN Departments d;

/* Type: SELF JOIN
   Logic: The table is joined with itself to compare rows within the same table.
   Context: We treat 'Employees' as two separate entities: 
            1. 'emp' (the worker)
            2. 'mgr' (the boss)
   Result: Lists the employee's name next to their manager's name.
*/
SELECT 
    emp.LastName AS Employee, 
    mgr.LastName AS Manager
FROM Employees emp
LEFT JOIN Employees mgr 
    ON emp.ManagerID = mgr.EmployeeID;


/* Type: NATURAL JOIN
   Logic: Automatically joins tables based on columns with the SAME NAME and compatible types.
   Risk: Highly risky in production! If a new common column is added later, the query logic changes silently.
*/

-- Notice: No 'ON' clause is needed here
SELECT LastName, DepartmentName
FROM Employees
NATURAL JOIN Departments;

/* Scenario: Analytics per Department
   Logic: 
   1. Use LEFT JOIN to include departments even if they have NO employees.
   2. Group the results by the Department Name.
   3. Calculate count and average salary.
*/

SELECT 
    d.DepartmentName,
    -- Count specific ID to avoid counting NULLs from the Left Join as 1
    COUNT(e.EmployeeID) AS TotalEmployees, 
    -- Calculate average, rounding to 2 decimal places
    ROUND(AVG(e.Salary), 2) AS AverageSalary
FROM Departments d
LEFT JOIN Employees e 
    ON d.DepartmentID = e.DepartmentID
GROUP BY d.DepartmentName
ORDER BY TotalEmployees DESC;
