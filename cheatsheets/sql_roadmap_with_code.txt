-- Example SQL Code Snippets --
-- Create a new database named 'CompanyDB'
CREATE DATABASE CompanyDB
WITH 
    OWNER = db_admin,          -- Specifies the owner of the database
    ENCODING = 'UTF8',         -- Sets the character encoding to UTF-8
    CONNECTION LIMIT = 25;     -- Limits the number of concurrent connections to 25;

CREATE TABLE Employees (
    EmployeeID INT AUTO_INCREMENT PRIMARY KEY, -- Unique identifier, auto-generated
    FirstName VARCHAR(50) NOT NULL,            -- This field cannot be empty
    LastName VARCHAR(50) NOT NULL,
    Email VARCHAR(100) UNIQUE,                 -- Ensures no duplicate email addresses
    BirthDate DATE,
    Salary DECIMAL(10, 2),                     -- Format for currency (e.g., 50000.00)
    Department VARCHAR(50) DEFAULT 'General'   -- Default value if none is provided
);

-- Create an index on the 'LastName' column to speed up search queries
CREATE INDEX idx_lastname 
ON Employees (LastName);

-- Create a view that shows only contact info (hides sensitive data like salary)
CREATE VIEW EmployeeContactList AS
SELECT FirstName, LastName, Email
FROM Employees;

-- Create or replace the procedure if it already exists to avoid errors (PostgreSQL syntax)
CREATE OR REPLACE PROCEDURE IncreaseSalary(
    p_EmployeeID INT, 
    p_Percentage DECIMAL
)
LANGUAGE plpgsql
AS $$
BEGIN
    -- Update the salary for the specific employee
    -- The calculation increases the current salary by the given percentage
    UPDATE Employees
    SET Salary = Salary * (1 + (p_Percentage / 100))
    WHERE EmployeeID = p_EmployeeID;

    -- Optional: Commit the transaction immediately (unique to Procedures in Postgres)
    COMMIT;
END;
$$;

-- Create or replace the procedure to allow overwriting without dropping it first (Oracle syntax)
CREATE OR REPLACE PROCEDURE IncreaseSalary (
    p_EmployeeID IN NUMBER,    -- Oracle prefers NUMBER instead of INT/DECIMAL
    p_Percentage IN NUMBER
)
AS
BEGIN
    -- Update the salary calculation
    -- Logic remains the same: Current Salary * (1 + Percentage/100)
    UPDATE Employees
    SET Salary = Salary * (1 + (p_Percentage / 100))
    WHERE EmployeeID = p_EmployeeID;

    -- In Oracle, you often commit explicitly to make changes permanent
    COMMIT;
END IncreaseSalary;
/

TRUNCATE OPERATIONS (TRUNCATE TABLE)
---------------------------------------------------------------------------------------------------------------
/* PostgreSQL Syntax */

-- 1. Simple Truncate: Clears data, keeps space allocated vaguely similar to before
TRUNCATE TABLE Logs;

-- 2. Reset Identity: Deletes data AND resets the auto-increment counter to 1
TRUNCATE TABLE Products RESTART IDENTITY;

-- 3. Cascade: The "Nuclear Option"
-- Deletes data in this table AND automatically deletes data in all tables 
-- that reference this table via Foreign Keys. Use with extreme caution!
TRUNCATE TABLE Departments CASCADE;

/* Oracle Syntax */

-- 1. Standard Truncate (implicitly performs DROP STORAGE)
-- Frees up the disk space for other tables to use
TRUNCATE TABLE Logs;

-- 2. Keep Space Allocated (Performance optimization)
-- Deletes rows but keeps the storage block reserved for this table.
-- Faster for subsequent INSERTs because no new space allocation is needed.
TRUNCATE TABLE Logs REUSE STORAGE;

-- 3. Cascade (Since Oracle 12c)
-- Recursively truncates child tables connected via Foreign Keys (ON DELETE CASCADE must be set)
TRUNCATE TABLE Departments CASCADE;

SELECT STATEMENT (SELECT, FROM, WHERE, GROUP BY, HAVING, ORDER BY, JOINS, SUBQUERIES, SET OPERATIONS)
------------------------------------------------------------------------------------------------------
SELECT FirstName, LastName, Salary, Department
FROM Employees
WHERE Department = 'IT' 
  AND Salary > 50000
ORDER BY Salary DESC; -- Sorts from highest to lowest

SELECT Department, AVG(Salary) as AvgSalary, COUNT(*) as EmployeeCount
FROM Employees
GROUP BY Department
HAVING AVG(Salary) > 60000; -- Filters groups with average salary over 60,000

SELECT 
    e.FirstName, 
    e.LastName, 
    d.DepartmentName
FROM Employees e                  -- 'e' is an alias for Employees
INNER JOIN Departments d          -- 'd' is an alias for Departments
    ON e.DepartmentID = d.DepartmentID;

 SELECT FirstName, Salary
FROM Employees
WHERE Salary > (
    SELECT AVG(Salary) FROM Employees
); -- Finds employees earning above average salary

-- Returns a list of ALL names from both tables
-- UNION removes duplicates, UNION ALL keeps them (and is faster)
SELECT LastName FROM Current_Employees
UNION ALL
SELECT LastName FROM Archived_Employees;

-- Who is in the archive BUT somehow still listed as active?
SELECT EmployeeID FROM Current_Employees
INTERSECT
SELECT EmployeeID FROM Archived_Employees;

/* Oracle specific */
SELECT ProductID FROM Inventory
MINUS
SELECT ProductID FROM Sales; -- Items that have never been sold

/* PostgreSQL specific */
SELECT ProductID FROM Inventory
EXCEPT
SELECT ProductID FROM Sales; -- Items that have never been sold

JOINS (INNER JOIN, LEFT JOIN, RIGHT JOIN, FULL OUTER JOIN, CROSS JOIN, SELF JOIN)
---------------------------------------------------------------------------------------------------------------

/* Type: INNER JOIN
   Logic: Returns only rows when there is a match in BOTH tables.
   Result: Employees without a department are excluded. 
           Departments with no employees are excluded.
*/
SELECT e.LastName, d.DepartmentName
FROM Employees e
INNER JOIN Departments d 
    ON e.DepartmentID = d.DepartmentID;

/* Type: LEFT OUTER JOIN
   Logic: Returns ALL records from the LEFT table (Employees), and the matched records from the right table.
   Result: If an employee has no department, they are still listed, but DepartmentName will be NULL.
*/
SELECT e.LastName, d.DepartmentName
FROM Employees e
LEFT JOIN Departments d 
    ON e.DepartmentID = d.DepartmentID;  

/* Type: RIGHT OUTER JOIN
   Logic: Returns ALL records from the RIGHT table (Departments), and the matched records from the left table.
   Result: If a department exists (e.g., 'Innovation Lab') but has no employees, it is listed.
           The employee columns (LastName) will be NULL for that row.
*/
SELECT e.LastName, d.DepartmentName
FROM Employees e
RIGHT JOIN Departments d 
    ON e.DepartmentID = d.DepartmentID;

/* Type: FULL OUTER JOIN
   Logic: Returns all records when there is a match in EITHER left or right table.
   Result: Combines the results of both LEFT and RIGHT joins.
           - You see employees without departments.
           - You see departments without employees.
           - NULLs appear wherever data is missing on the other side.
*/
SELECT e.LastName, d.DepartmentName
FROM Employees e
FULL OUTER JOIN Departments d 
    ON e.DepartmentID = d.DepartmentID;

/* Type: CROSS JOIN (Cartesian Product)
   Logic: Returns the combination of EVERY row from the first table with EVERY row from the second table.
   Result: If Employees has 100 rows and Departments has 10 rows, the result is 1,000 rows.
   Warning: Can be very slow on large tables!
*/
SELECT e.LastName, d.DepartmentName
FROM Employees e
CROSS JOIN Departments d;

/* Type: SELF JOIN
   Logic: The table is joined with itself to compare rows within the same table.
   Context: We treat 'Employees' as two separate entities: 
            1. 'emp' (the worker)
            2. 'mgr' (the boss)
   Result: Lists the employee's name next to their manager's name.
*/
SELECT 
    emp.LastName AS Employee, 
    mgr.LastName AS Manager
FROM Employees emp
LEFT JOIN Employees mgr 
    ON emp.ManagerID = mgr.EmployeeID;


/* Type: NATURAL JOIN
   Logic: Automatically joins tables based on columns with the SAME NAME and compatible types.
   Risk: Highly risky in production! If a new common column is added later, the query logic changes silently.
*/

-- Notice: No 'ON' clause is needed here
SELECT LastName, DepartmentName
FROM Employees
NATURAL JOIN Departments;

/* Scenario: Analytics per Department
   Logic: 
   1. Use LEFT JOIN to include departments even if they have NO employees.
   2. Group the results by the Department Name.
   3. Calculate count and average salary.
*/

SELECT 
    d.DepartmentName,
    -- Count specific ID to avoid counting NULLs from the Left Join as 1
    COUNT(e.EmployeeID) AS TotalEmployees, 
    -- Calculate average, rounding to 2 decimal places
    ROUND(AVG(e.Salary), 2) AS AverageSalary
FROM Departments d
LEFT JOIN Employees e 
    ON d.DepartmentID = e.DepartmentID
GROUP BY d.DepartmentName
ORDER BY TotalEmployees DESC;

INSERT STATEMENT (INSERT INTO, INSERT SELECT, BULK INSERT)
---------------------------------------------------------------------------------------------------------------

/* Standard SQL (Works in Postgres & Oracle)
   Logic: Adds a single new record to the Employees table.
   Note: If columns are omitted (like EmployeeID), the DB tries to use auto-increment or default values.
*/
INSERT INTO Employees (FirstName, LastName, Email, Salary, DepartmentID)
VALUES ('Alice', 'Wonderland', 'alice@example.com', 55000.00, 10);

-- Oracle Requirement:
COMMIT; -- Don't forget this in Oracle!

/* PostgreSQL Syntax
   Logic: Insert multiple rows in a single command. Very efficient.
*/
INSERT INTO Employees (FirstName, LastName, Salary)
VALUES 
    ('Bob', 'Builder', 45000.00),
    ('Charlie', 'Chocolate', 62000.00),
    ('Dora', 'Explorer', 48000.00);


/* Oracle Syntax (Classic)
   Logic: Insert into multiple rows at once. Requires a dummy select at the end.
*/
INSERT ALL
    INTO Employees (FirstName, LastName, Salary) VALUES ('Bob', 'Builder', 45000)
    INTO Employees (FirstName, LastName, Salary) VALUES ('Charlie', 'Choc', 62000)
SELECT * FROM dual;   

/* Type: INSERT INTO ... SELECT
   Logic: Selects data from one table and inserts it directly into another.
   Prerequisite: The target table (Project_Team) must already exist.
*/
INSERT INTO Project_Team (MemberName, MemberRole)
SELECT LastName, 'Developer'
FROM Employees
WHERE DepartmentID = (SELECT DepartmentID FROM Departments WHERE DepartmentName = 'IT');

/* PostgreSQL Specific: COPY
   Logic: Reads a CSV file from the server's disk and fills the table.
   Note: Requires superuser privileges or specific file access.
*/
COPY Employees(FirstName, LastName, Email, Salary)
FROM '/path/to/new_hires.csv'
DELIMITER ','
CSV HEADER;

/* Oracle Approach: INSERT from External Table
   Logic: 1. Create a link to the CSV file (External Table).
          2. Use INSERT SELECT to load it.
*/
INSERT INTO Employees (FirstName, LastName, Salary)
SELECT col1, col2, col3
FROM My_CSV_External_Table;

UPDATE STATEMENT (UPDATE, SET, WHERE, JOINED UPDATE)
---------------------------------------------------------------------------------------------------------------
/* Standard UPDATE
   Logic: Increases the salary by 5000 for a specific employee.
   Safety: Always use a WHERE clause! Without it, EVERY row gets updated.
*/
UPDATE Employees
SET Salary = Salary + 5000
WHERE EmployeeID = 101;

/* PostgreSQL: Update with Join
   Logic: Uses the 'FROM' clause to join the source of the data.
*/
UPDATE Employees e
SET Salary = e.Salary + b.BonusAmount
FROM Bonuses b
WHERE e.EmployeeID = b.EmployeeID;

/* Oracle: Update with Correlated Subquery
   Logic: We look up the value for each row individually.
*/
UPDATE Employees e
SET Salary = e.Salary + (
    SELECT b.BonusAmount 
    FROM Bonuses b 
    WHERE b.EmployeeID = e.EmployeeID
)
WHERE EXISTS ( -- Critical: Only update rows that actually have a bonus!
    SELECT 1 
    FROM Bonuses b 
    WHERE b.EmployeeID = e.EmployeeID
);

DELETE STATEMENT (DELETE FROM, WHERE, JOINED DELETE)
-----------------------------------------------------------------------------------------------
/* PostgreSQL: Delete using another table
   Logic: The 'USING' keyword allows referencing other tables for the condition.
*/
DELETE FROM Employees
USING ClosedDepartments
WHERE Employees.DepartmentID = ClosedDepartments.DepartmentID;

/* Oracle: Delete with Subquery
   Logic: Delete rows where the ID is found in the list of closed departments.
*/
DELETE FROM Employees
WHERE DepartmentID IN (
    SELECT DepartmentID 
    FROM ClosedDepartments
);

MERGE STATEMENT (MERGE INTO, USING, ON, WHEN MATCHED, WHEN NOT MATCHED)
---------------------------------------------------------------------------------------------
/* MERGE Statement (Works in Oracle & PostgreSQL 15+)
   Target: The table receiving changes (Target_Products)
   Source: The incoming data (Source_Products)
*/
MERGE INTO Target_Products t
USING Source_Products s
ON (t.ProductID = s.ProductID) -- The matching condition (Primary Key)

-- 1. If the ID exists -> Update the price and name
WHEN MATCHED THEN
    UPDATE SET 
        t.ProductName = s.ProductName,
        t.Price = s.Price

-- 2. If the ID does NOT exist -> Insert a new row
WHEN NOT MATCHED THEN
    INSERT (ProductID, ProductName, Price)
    VALUES (s.ProductID, s.ProductName, s.Price);

LOCK TABLE STATEMENT (LOCK TABLE, LOCK MODE)
-------------------------------------------------------------------------------------------------

 /* PostgreSQL Locking
   Mode: ACCESS EXCLUSIVE (Most restrictive lock in Postgres).
   Effect: Nobody can read or write to the table until transaction ends.
*/
BEGIN; -- Start transaction
    LOCK TABLE Employees IN ACCESS EXCLUSIVE MODE;
    
    -- Do critical maintenance here...
    
COMMIT; -- Release lock

/* Oracle Locking
   Mode: EXCLUSIVE.
   Effect: Allows queries (SELECT) from others, but blocks any INSERT/UPDATE/DELETE/LOCK.
   Option: NOWAIT (Returns error immediately if table is already locked by someone else).
*/
LOCK TABLE Employees IN EXCLUSIVE MODE NOWAIT;

-- Lock is released automatically on COMMIT or ROLLBACK

GRANT STATEMENT (GRANT PRIVILEGES, ON OBJECT, TO USER/ROLE)
-------------------------------------------------------------------------------------------------
/* Standard SQL (Postgres & Oracle)
   Object: The 'Employees' table.
   Privileges: User can READ (Select) and ADD (Insert) data, but NOT delete or update.
   User: 'junior_dev'
*/
GRANT SELECT, INSERT 
ON Employees 
TO junior_dev;

/* Oracle: Granting execution rights on a procedure */
GRANT EXECUTE ON IncreaseSalary TO manager_role;

/* PostgreSQL Specifics */

-- 1. Allow running the procedure
GRANT EXECUTE ON PROCEDURE IncreaseSalary TO manager_role;

-- 2. CRITICAL: If inserting data, user needs rights on the sequence too!
-- Otherwise they get "Permission denied for sequence..." errors.
GRANT USAGE, SELECT ON SEQUENCE employees_id_seq TO junior_dev;

/* Oracle System Privileges
   CREATE SESSION: Allows the user to connect to the DB.
   CREATE TABLE: Allows user to create their own tables.
*/
GRANT CREATE SESSION TO new_user;
GRANT CREATE TABLE TO new_user;

/* RBAC */

-- Create a generic role (container for privileges)
CREATE ROLE read_only_staff;

-- Give read access to the role
GRANT SELECT ON Employees TO read_only_staff;
GRANT SELECT ON Departments TO read_only_staff;

/* Grant the role to the actual user
   Now 'alice' inherits all permissions from 'read_only_staff'.
*/
GRANT read_only_staff TO alice;



/* WITH GRANT OPTION
   Logic: 'manager' gets SELECT rights AND the power to grant SELECT to others.
   Warning: If you revoke rights from 'manager' later, it might cascade to others!
*/
GRANT SELECT, UPDATE 
ON Employees 
TO manager 
WITH GRANT OPTION;

/* PostgreSQL only: Bulk Grant
   Logic: Grants SELECT on ALL tables currently in the schema.
   Note: Does not automatically apply to tables created in the future 
         (unless you use ALTER DEFAULT PRIVILEGES).
*/
GRANT SELECT ON ALL TABLES IN SCHEMA public TO read_only_staff;

SAVEPOINT STATEMENT (SAVEPOINT NAME)
---------------------------------------------------------------------------------------------------
/* Standard SQL (Postgres & Oracle)
   Action: Remove the DELETE privilege on the Employees table.
   Result: The user can still SELECT or INSERT (if previously granted), 
           but gets a "permission denied" error when trying to delete.
*/
REVOKE DELETE 
ON Employees 
FROM junior_dev;

/* Revoking a Role
   Action: Alice loses all privileges associated with the 'manager_role'.
*/
REVOKE manager_role FROM alice;
/* PostgreSQL Specific: CASCADE
   Logic: Revoke rights from 'manager' AND automatically revoke rights 
          from anyone the manager granted rights to.
*/
REVOKE SELECT ON Employees FROM manager CASCADE;

/* Scenario: Successful Transfer */

-- 1. Start Transaction (Essential in Postgres, implicit in Oracle)
BEGIN; 

-- 2. Deduct 500 from Account A
UPDATE Accounts SET Balance = Balance - 500 WHERE AccountID = 'A';

-- 3. Add 500 to Account B
UPDATE Accounts SET Balance = Balance + 500 WHERE AccountID = 'B';

-- 4. Save changes permanently
/* If power fails before this line, NO money changes hands. */
COMMIT;

/* Scenario: Something goes wrong */

BEGIN; 

-- 1. Deduct money from A
UPDATE Accounts SET Balance = Balance - 500 WHERE AccountID = 'A';

-- 2. Simulate an error (e.g., Account B does not exist or server crash)
-- Oops! We cannot credit Account B. We must not leave Account A with less money!

-- 3. Undo everything in this transaction block
ROLLBACK;

/* Result: Account A has its original balance back. */

/* Scenario: Partial Rollback using Savepoints */

BEGIN;

-- Step 1: Insert Product 1 (Safe)
INSERT INTO Products (Name) VALUES ('Laptop');

-- Step 2: Insert Product 2 (Safe)
INSERT INTO Products (Name) VALUES ('Mouse');

-- Set a marker here
SAVEPOINT my_checkpoint;

-- Step 3: Try to insert Product 3 (Risky - maybe duplicate ID?)
INSERT INTO Products (Name) VALUES ('Keyboard');

-- Let's assume Step 3 caused an error. 
-- Instead of cancelling the Laptop and Mouse, we go back to the marker:
ROLLBACK TO SAVEPOINT my_checkpoint;

-- Step 4: Save the transaction
COMMIT;

/* Result: 
   - Laptop: SAVED
   - Mouse: SAVED
   - Keyboard: NOT SAVED (Rolled back)
*/

SET TRANSACTION STATEMENT (SET TRANSACTION ISOLATION LEVEL, READ WRITE/ONLY)
-------------------------------------------------------------------------------------------------
/* PostgreSQL & Oracle Syntax */
SET TRANSACTION READ ONLY;

-- Now, this is safe:
SELECT * FROM Employees;

-- BUT this will trigger an error immediately:
DELETE FROM Employees; -- Error: cannot execute DELETE in a read-only transaction

SET TRANSACTION READ WRITE;

/* Standard Mode */
SET TRANSACTION ISOLATION LEVEL READ COMMITTED;

/* Consistent Reporting Mode */
SET TRANSACTION ISOLATION LEVEL REPEATABLE READ;

/* Maximum Safety Mode */
SET TRANSACTION ISOLATION LEVEL SERIALIZABLE;

/* Code for the Finance Report */
BEGIN;

-- Lock the view of the data to the moment this command runs
SET TRANSACTION ISOLATION LEVEL REPEATABLE READ;
SET TRANSACTION READ ONLY; -- Good practice for reports

-- Run complex queries...
SELECT SUM(Balance) FROM Accounts;
-- ... take coffee break ...
-- ... run more queries ...

COMMIT; -- End the snapshot
