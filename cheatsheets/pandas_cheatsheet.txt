# Use .copy() when creating a new DataFrame from an existing one to avoid SettingWithCopyWarning
df = df.copy()  # Create a copy of the DataFrame

# Chain operations with method chaining
df = (df.dropna()
        .rename(columns={'old_name': 'new_name'})
        .sort_values(by='column', ascending=False))  # Chain operations

# Use .loc[] for label-based indexing and .iloc[] for position-based indexing
df.loc[0, 'column']  # Access value by label

# Set dtype='category' for categorical data to save memory and improve performance
df['column'] = df['column'].astype('category')  # Convert column to categorical

# Use inplace=True to modify the DataFrame in place without creating a new object
df.dropna(inplace=True)  # Remove missing values in place

# Use .query() for complex filtering conditions
df.query('column > 10 and column < 20')  # Filter rows using query

# Use .groupby() for grouping data and performing aggregations
df.groupby('column').agg({'column2': 'mean'})  # Group by and aggregate

# Use .assign() to add new columns or modify existing ones
df = df.assign(new_column=df['column'] * 2)  # Add new column
