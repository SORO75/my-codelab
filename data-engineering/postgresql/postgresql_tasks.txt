# Postgres installation on Ubuntu
# update repository and install PostgreSQL
        sudo apt update
        sudo apt upgrade
        sudo apt install -y postgresql postgresql-contrib

# check PostgreSQL service status
        sudo systemctl status postgresql
# start PostgreSQL
        sudo systemctl start postgresql
# restart PostgreSQL
        sudo systemctl restart postgresql       
# enable it to start on boot
        sudo systemctl enable postgresql    
# verifiy PostgreSQL installation
        psql --version 


# switch to the postgres account
        sudo -i -u postgres
# open the PostgreSQL shell
        psql
# active Connections
        \conninfo

# set password for PostgreSQL admin user
       ALTER USER postgres WITH PASSWORD 'securePassword';

# create a new database 
        CREATE DATABASE my_database;
    # bash
        sudo -u postgres createdb my_database  

# create a new user und grant privileges
        CREATE USER my_user WITH ENCRYPTED PASSWORD 'strongPassword';
        GRANT ALL PRIVILEGES ON DATABASE my_database TO my_user;


# list all databases
        \l 
# connect to a specific database
        \c my_database

# list all users
        \du 

# exit psql 
        \q or exit

# volume mounting for data persistence
docker run --name postgres-testcontainer -e POSTGRES_USER=myuser -e POSTGRES_PASSWORD=mypassword -e POSTGRES_DB=mydb -p 5454:5432 -v pgdata:/var/lib/postgresql/data -d postgres:17-alpine

#Backup and restore
# backup a database
docker exec -t postgres-testcontainer pg_dump -U myuser testdb > backup.sql
# restore a database
cat backup.sql | docker exec -i postgres-testcontainer psql -U myuser -d testdb
